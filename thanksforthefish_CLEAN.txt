import matplotlib.pyplot as plt
import heapq
import time

def get_primes(n):
    out =[]
    k = 2
    while len(out) < n:
        if all(k % d for d in range(2, int(k**0.5) + 1)):
            out.append(k)
        k += 1
    return np.array(out)

def build_adelic_lattice(parent_seeds, max_composites=2000):
    base_freqs = np.log(parent_seeds)
    heap = [(0.0, 0)]
    seen = {0.0}
    composites =[]
    
    while len(composites) < max_composites and heap:
        freq, max_idx = heapq.heappop(heap)
        composites.append(freq)
        for i in range(max_idx, len(base_freqs)):
            new_freq = freq + base_freqs[i]
            q = round(new_freq, 7)
            if q not in seen:
                seen.add(q)
                heapq.heappush(heap, (new_freq, i))
                
    freqs = np.array(composites[1:])
    amps = np.exp(-freqs / 2.0)
    return freqs, amps

def fast_forward_to_level_2():

    # Level 0
    l0_seeds = get_primes(40)

    # Level 1 Scars
    mag0 = np.abs(z0)
    min_idx0 = np.where((mag0[1:-1] < mag0[:-2]) & (mag0[1:-1] < mag0[2:]))[0] + 1
    l1_seeds = t_arr[min_idx0[mag0[min_idx0] < 0.7 * np.mean(mag0)]][:60]

    min_idx1 = np.where((mag1[1:-1] < mag1[:-2]) & (mag1[1:-1] < mag1[2:]))[0] + 1

    mag2 = np.abs(z2)
    min_idx = np.where((mag2[1:-1] < mag2[:-2]) & (mag2[1:-1] < mag2[2:]))[0] + 1
    l_mean = np.convolve(mag2, np.ones(50)/50, mode='same')
    scars = t_arr[min_idx[mag2[min_idx] < 0.7 * l_mean[min_idx]]]
    
    spacings = np.diff(scars)
    window = min(20, len(spacings) // 2)
    unfolded =[]
    for i in range(len(spacings)):
        start, end = max(0, i - window), min(len(spacings), i + window)
        local_mean = np.mean(spacings[start:end])
        if local_mean > 0: unfolded.append(spacings[i] / local_mean)
    
    unfolded = np.array(unfolded)
    
    # --- EXPERIMENT 2: THERMODYNAMICS & THE BIG BANG ---
    # Partition Function Z(T) = sum(e^(-E_i / T))
    # Heat Capacity C_v(T) = d<E>/dT
    temperatures = np.linspace(0.1, 3.0, 300)
    heat_capacity =[]
    
    for T in temperatures:
        beta = 1.0 / T
        Z = np.sum(np.exp(-beta * freqs))

        Cv = (E_sq_mean - E_mean**2) / (T**2)
        heat_capacity.append(Cv)

    ax.plot(s_vals, gue_curve, color='#ff5555', lw=2, label="Theoretical GUE")
    ax.set_title("Quantum Mechanics: Level Repulsion", color='#ccc')
    ax.legend()
    ax.tick_params(colors='#666')
    
    # Plot 2: Heat Capacity (The Big Bang)
    ax = axs[1]

    ax.tick_params(colors='#666')

    ax.plot(t_arr[:1000], (mag2[:1000])**2, color='#5a8a6a', lw=1, alpha=0.7)
    ax.axhline(vacuum_energy, color='#8a4a4a', linestyle='--', label=f"Vacuum Energy Background: {vacuum_energy:.2f}")
    ax.set_title("Cosmology: Dark Energy Background", color='#ccc')
    ax.legend()

    print("\n--- LEVEL 2 PHYSICS REPORT ---")
